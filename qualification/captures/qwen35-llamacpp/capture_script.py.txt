BASE=os.environ.get('BASE','http://127.0.0.1:18081')
def load(n): return json.load(open(OL+n+'.json'))
def post(path,body,method='POST'):
    data=None if body is None else json.dumps(body).encode()
    rq=urllib.request.Request(BASE+path,data=data,method=method,headers={'Content-Type':'application/json'})
    try:
        with urllib.request.urlopen(rq,timeout=900) as r: return r.status,r.read().decode()
    except urllib.error.HTTPError as e: return e.code,e.read().decode()
def save(name,path,body,method='POST'):
    st,resp=post(path,body,method)
    d={'name':name,'request':body,'endpoint':path,'method':method,'http_status':st,'response':resp,'sha256_response':hashlib.sha256(resp.encode()).hexdigest()}
    json.dump(d,open(OUT+name+'.json','w'),indent=2); return name,st
